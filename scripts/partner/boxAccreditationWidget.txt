<script>
(function(){
    var form = document.getElementById('AccWidgetForm');
    if (!form) return;

    document.getElementById('AccWidgetOpenAt').value = String(Date.now());

    var phone = document.getElementById('AccWidgetPhone');
    phone.addEventListener('input', function(e){
        var v = e.target.value.replace(/\D/g, '');
        if (v.startsWith('8')) v = '7' + v.slice(1);
        if (v && !v.startsWith('7')) v = '7' + v;
        v = v.slice(0, 11);
        var f = '+7';
        if (v.length > 1) f += ' (' + v.slice(1, 4);
        if (v.length >= 4) f += ') ' + v.slice(4, 7);
        if (v.length >= 7) f += '-' + v.slice(7, 9);
        if (v.length >= 9) f += '-' + v.slice(9, 11);
        e.target.value = f;
    });

    form.addEventListener('submit', function(e){
        e.preventDefault();
        if (form.website.value) return; // honeypot
        if (phone.value.replace(/\D/g, '').length < 11) {
            showError('Укажите телефон полностью');
            return;
        }
        if (!form.consent.checked) {
            showError('Подтвердите согласие на обработку персональных данных');
            return;
        }

        var btn = document.getElementById('AccWidgetSubmit');
        btn.disabled = true;
        btn.textContent = 'Отправка...';

        var fd = new FormData(form);
        var accWhen = fd.get('acc_when') || 'не указан';
        fd.set('form_source', fd.get('form_source') + ' | срок: ' + accWhen);
        // дублируем срок в message чтобы менеджер видел в B24
        fd.set('message', 'Срок следующей аккредитации: ' + accWhen);

        fetch('/ajax-lead/', { method: 'POST', body: fd })
            .then(function(r){ return r.json(); })
            .then(function(j){
                if (j.ok) {
                    document.querySelector('.acc-widget__form-wrap').hidden = true;
                    document.getElementById('AccWidgetSuccess').hidden = false;
                    if (typeof ym === 'function') ym(75081295, 'reachGoal', 'acc_widget_submit');
                } else {
                    showError('Ошибка отправки. Попробуйте ещё раз.');
                    btn.disabled = false;
                    btn.textContent = 'Подобрать программу';
                }
            })
            .catch(function(){
                showError('Ошибка соединения');
                btn.disabled = false;
                btn.textContent = 'Подобрать программу';
            });
    });

    function showError(msg){
        var el = document.getElementById('AccWidgetError');
        el.textContent = msg;
        el.hidden = false;
    }
})();
</script>
