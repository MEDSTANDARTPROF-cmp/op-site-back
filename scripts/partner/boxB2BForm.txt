<script>
(function(){
    var form = document.getElementById('B2BBlockForm');
    if (!form) return;

    document.getElementById('B2BBlockOpenAt').value = String(Date.now());

    // Phone mask
    var phone = document.getElementById('B2BPhone');
    phone.addEventListener('input', function(e){
        var v = e.target.value.replace(/\D/g, '');
        if (v.startsWith('8')) v = '7' + v.slice(1);
        if (v && !v.startsWith('7')) v = '7' + v;
        v = v.slice(0, 11);
        var f = '+7';
        if (v.length > 1) f += ' (' + v.slice(1, 4);
        if (v.length >= 4) f += ') ' + v.slice(4, 7);
        if (v.length >= 7) f += '-' + v.slice(7, 9);
        if (v.length >= 9) f += '-' + v.slice(9, 11);
        e.target.value = f;
    });

    // INN: only digits
    var inn = document.getElementById('B2BInn');
    inn.addEventListener('input', function(e){
        e.target.value = e.target.value.replace(/\D/g, '').slice(0, 12);
    });

    function showError(msg){
        var el = document.getElementById('B2BError');
        el.textContent = msg;
        el.hidden = false;
    }

    form.addEventListener('submit', function(e){
        e.preventDefault();
        if (form.website.value) return; // honeypot

        var innVal = inn.value.trim();
        if (innVal.length !== 10 && innVal.length !== 12) {
            showError('ИНН должен содержать 10 или 12 цифр');
            return;
        }
        if (!form.employees.value) {
            showError('Укажите количество сотрудников');
            return;
        }
        if (phone.value.replace(/\D/g, '').length < 11) {
            showError('Укажите телефон полностью');
            return;
        }
        if (!form.consent.checked) {
            showError('Подтвердите согласие на обработку персональных данных');
            return;
        }

        var btn = document.getElementById('B2BSubmit');
        btn.disabled = true;
        var btnText = btn.querySelector('span');
        var oldTxt = btnText.textContent;
        btnText.textContent = 'Отправка...';

        var fd = new FormData(form);
        fetch('/ajax-lead', { method: 'POST', body: fd })
            .then(function(r){ return r.json(); })
            .then(function(j){
                if (j.ok) {
                    document.querySelector('.b2b-block__form-wrap').hidden = true;
                    document.getElementById('B2BSuccess').hidden = false;
                    if (typeof ym === 'function') ym(75081295, 'reachGoal', 'b2b_form_submit');
                } else {
                    showError('Ошибка отправки: ' + (j.error || 'попробуйте позже'));
                    btn.disabled = false;
                    btnText.textContent = oldTxt;
                }
            })
            .catch(function(){
                showError('Ошибка соединения');
                btn.disabled = false;
                btnText.textContent = oldTxt;
            });
    });
})();
</script>
